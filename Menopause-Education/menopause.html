def assess_cultural_perspective(responses):
    """
    Analyze user responses to determine cultural pathway
    Returns: 'liberation', 'medical', or 'neutral'
    """
    scores = {"liberation": 0, "medical": 0, "neutral": 0}
    
    for question_id, response in responses.items():
        for pathway, indicators in cultural_assessment["scoring_weights"].items():
            pathway_name = pathway.replace("_indicators", "")
            if response in indicators:
                scores[pathway_name] += 1
    
    return max(scores, key=scores.get)

class ContentRouter:
    def __init__(self):
        self.pathways = {
            "liberation": LiberationPathway(),
            "medical": MedicalPathway(), 
            "neutral": NeutralPathway()
        }
    
    def route_user(self, assessment_results):
        """Route user to appropriate content pathway based on cultural assessment"""
        score = self.calculate_pathway_score(assessment_results)
        return self.pathways[score["dominant_pathway"]]
    
    def calculate_pathway_score(self, results):
        """Calculate which pathway best matches user's cultural perspective"""
        pathway_scores = {
            "liberation": 0,
            "medical": 0,
            "neutral": 0
        }
        
        # Score based on response patterns
        for response in results:
            if response in ["liberation", "empowerment", "traditional"]:
                pathway_scores["liberation"] += 1
            elif response in ["medical", "treatment", "professional"]:
                pathway_scores["medical"] += 1
            else:
                pathway_scores["neutral"] += 1
        
        dominant_pathway = max(pathway_scores, key=pathway_scores.get)
        confidence = pathway_scores[dominant_pathway] / sum(pathway_scores.values())
        
        return {
            "dominant_pathway": dominant_pathway,
            "confidence": confidence,
            "all_scores": pathway_scores
        }

class SafetyValidator:
    def __init__(self):
        self.medical_keywords = self.load_medical_keywords()
        self.safety_thresholds = {
            "confidence_min": 0.8,
            "medical_accuracy": 0.95,
            "cultural_sensitivity": 0.9
        }
        self.restricted_topics = [
            "specific_medical_diagnosis",
            "medication_recommendations",
            "emergency_medical_situations"
        ]
    
    def validate_response(self, response, user_context):
        """Multi-layer safety validation"""
        medical_check = self.verify_medical_accuracy(response)
        bias_check = self.detect_cultural_bias(response, user_context)
        confidence_check = self.assess_confidence(response)
        content_check = self.validate_content_appropriateness(response)
        
        return {
            "approved": all([medical_check, bias_check, confidence_check, content_check]),
            "requires_human_review": not medical_check or not content_check,
            "escalate_to_expert": confidence_check < self.safety_thresholds["confidence_min"],
            "safety_flags": self.get_safety_flags(response),
            "recommendations": self.get_improvement_recommendations(response)
        }
    
    def verify_medical_accuracy(self, response):
        """Validate medical information against established sources"""
        # Check against medical fact database
        # Verify claims don't contradict established medical knowledge
        # Flag any absolute medical statements without proper caveats
        medical_claims = self.extract_medical_claims(response)
        
        for claim in medical_claims:
            if not self.verify_against_medical_database(claim):
                return False
        return True
    
    def validate_content_appropriateness(self, response):
        """Ensure content doesn't venture into restricted medical territory"""
        for topic in self.restricted_topics:
            if self.contains_restricted_content(response, topic):
                return False
        return True
    
    def load_medical_keywords(self):
        """Load validated medical terminology and fact database"""
        return {
            "symptoms": ["hot_flashes", "night_sweats", "mood_changes", "sleep_disruption"],
            "treatments": ["hormone_therapy", "lifestyle_changes", "alternative_treatments"],
            "safety_phrases": ["consult_healthcare_provider", "individual_results_vary", "not_medical_advice"]
        }
    
    def assess_confidence(self, response):
        """Assess AI confidence level in response"""
        uncertainty_indicators = ["might", "could", "possibly", "sometimes", "may vary"]
        certainty_indicators = ["will", "always", "never", "definitely", "guaranteed"]
        
        uncertainty_count = sum(1 for indicator in uncertainty_indicators if indicator in response.lower())
        certainty_count = sum(1 for indicator in certainty_indicators if indicator in response.lower())
        
        # Higher uncertainty_count relative to certainty_count = higher confidence in safety
        if certainty_count > uncertainty_count:
            return 0.6  # Low confidence due to overconfident language
        else:
            return 0.9  # High confidence due to appropriate uncertainty
    
    def get_safety_flags(self, response):
        """Identify specific safety concerns in response"""
        flags = []
        
        if "definitely" in response.lower() or "guaranteed" in response.lower():
            flags.append("overconfident_medical_claims")
        
        if not any(phrase in response.lower() for phrase in ["consult", "healthcare provider", "doctor"]):
            flags.append("missing_medical_disclaimer")
        
        if any(word in response.lower() for word in ["cure", "treatment", "medication"]):
            flags.append("potential_medical_advice")
        
        return flags

def detect_cultural_bias(response_text, user_cultural_context):
    """
    Detect potential cultural bias in AI responses
    Returns bias score and specific bias indicators found
    """
    bias_indicators = {
        "western_centrism": {
            "keywords": ["should", "must", "always", "never", "normal", "standard"],
            "weight": 0.8,
            "description": "Imposing universal standards without cultural context"
        },
        "medical_bias": {
            "keywords": ["abnormal", "disorder", "problem", "failure", "dysfunction", "deficiency"],
            "weight": 0.9,
            "description": "Pathologizing natural menopause experiences"
        },
        "age_bias": {
            "keywords": ["old", "elderly", "declining", "deteriorating", "past_prime"],
            "weight": 0.7,
            "description": "Negative framing of aging and menopause"
        },
        "gender_bias": {
            "keywords": ["emotional", "irrational", "moody", "hysterical"],
            "weight": 0.9,
            "description": "Stereotypical gender assumptions"
        },
        "socioeconomic_bias": {
            "keywords": ["expensive", "luxury", "premium", "high_end"],
            "weight": 0.6,
            "description": "Assuming access to expensive treatments"
        }
    }
    
    cultural_appropriateness = {
        "liberation_pathway": check_empowerment_language(response_text),
        "medical_pathway": check_supportive_language(response_text),
        "neutral_pathway": check_balanced_language(response_text)
    }
    
    bias_score = calculate_bias_score(response_text, bias_indicators, cultural_appropriateness)
    
    return {
        "overall_bias_score": bias_score,
        "bias_indicators_found": get_bias_indicators_found(response_text, bias_indicators),
        "cultural_appropriateness_score": cultural_appropriateness[user_cultural_context],
        "recommendations": get_bias_reduction_recommendations(response_text, bias_indicators)

def check_empowerment_language(text):
    """Check for empowering, positive language appropriate for liberation pathway"""
    empowerment_keywords = [
        "empowerment", "wisdom", "freedom", "liberation", "celebration", 
        "growth", "transformation", "strength", "resilience", "opportunity"
    ]
    
    negative_keywords = [
        "loss", "decline", "failure", "problem", "abnormal", "disorder"
    ]
    
    empowerment_count = sum(1 for keyword in empowerment_keywords if keyword in text.lower())
    negative_count = sum(1 for keyword in negative_keywords if keyword in text.lower())
    
    # Higher empowerment and lower negative = better score
    total_keywords = empowerment_count + negative_count
    if total_keywords == 0:
        return 0.7  # Neutral
    
    return empowerment_count / total_keywords

def check_supportive_language(text):
    """Check for supportive, medically-informed language for medical pathway"""
    supportive_keywords = [
        "support", "management", "treatment", "care", "help", "relief", 
        "effective", "evidence-based", "professional", "guidance"
    ]
    
    dismissive_keywords = [
        "just", "simply", "only", "minor", "insignificant", "imaginary"
    ]
    
    supportive_count = sum(1 for keyword in supportive_keywords if keyword in text.lower())
    dismissive_count = sum(1 for keyword in dismissive_keywords if keyword in text.lower())
    
    # Higher supportive and lower dismissive = better score
    total_keywords = supportive_count + dismissive_count
    if total_keywords == 0:
        return 0.7  # Neutral
    
    return supportive_count / total_keywords

def check_balanced_language(text):
    """Check for balanced, informative language for neutral pathway"""
    balanced_keywords = [
        "varies", "individual", "different", "options", "consider", 
        "depends", "may", "can", "sometimes", "often"
    ]
    
    absolute_keywords = [
        "always", "never", "all", "none", "everyone", "nobody", "definitely"
    ]
    
    balanced_count = sum(1 for keyword in balanced_keywords if keyword in text.lower())
    absolute_count = sum(1 for keyword in absolute_keywords if keyword in text.lower())
    
    # Higher balanced and lower absolute = better score
    total_keywords = balanced_count + absolute_count
    if total_keywords == 0:
        return 0.7  # Neutral
    
    return balanced_count / total_keywords

def calculate_bias_score(text, bias_indicators, cultural_scores):
    """Calculate overall bias score combining multiple factors"""
    bias_score = 0
    total_weight = 0
    
    for bias_type, config in bias_indicators.items():
        found_indicators = sum(1 for keyword in config["keywords"] if keyword in text.lower())
        if found_indicators > 0:
            bias_score += found_indicators * config["weight"]
        total_weight += config["weight"]
    
    # Normalize bias score (lower is better)
    normalized_bias = bias_score / total_weight if total_weight > 0 else 0
    
    # Factor in cultural appropriateness (higher is better)
    avg_cultural_score = sum(cultural_scores.values()) / len(cultural_scores)
    
    # Final score: low bias + high cultural appropriateness = good score
    final_score = (1 - normalized_bias) * 0.6 + avg_cultural_score * 0.4
    
    return final_score

def get_bias_indicators_found(text, bias_indicators):
    """Return specific bias indicators found in the text"""
    found = {}
    for bias_type, config in bias_indicators.items():
        found_keywords = [keyword for keyword in config["keywords"] if keyword in text.lower()]
        if found_keywords:
            found[bias_type] = {
                "keywords_found": found_keywords,
                "description": config["description"]
            }
    return found

def get_bias_reduction_recommendations(text, bias_indicators):
    """Provide specific recommendations to reduce identified biases"""
    recommendations = []
    
    # Check for each type of bias and provide specific recommendations
    if any(keyword in text.lower() for keyword in bias_indicators["western_centrism"]["keywords"]):
        recommendations.append("Consider adding cultural context and acknowledging diverse perspectives on menopause")
    
    if any(keyword in text.lower() for keyword in bias_indicators["medical_bias"]["keywords"]):
        recommendations.append("Frame menopause as a natural transition rather than a medical problem")
    
    if any(keyword in text.lower() for keyword in bias_indicators["age_bias"]["keywords"]):
        recommendations.append("Use positive or neutral language about aging and life transitions")
    
    if any(keyword in text.lower() for keyword in bias_indicators["gender_bias"]["keywords"]):
        recommendations.append("Avoid stereotypical assumptions about women's emotional responses")
    
    if any(keyword in text.lower() for keyword in bias_indicators["socioeconomic_bias"]["keywords"]):
        recommendations.append("Include accessible options and acknowledge economic constraints")
    
    return recommendations
